cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
project(fengshen_llama_delivery LANGUAGES C CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This locked delivery requires MSVC x64")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_executable(llama-runtime component_info.cpp)
target_compile_features(llama-runtime PRIVATE cxx_std_17)
# Metadata must remain runnable without AVX, regardless of the server profile.
target_compile_options(llama-runtime PRIVATE /arch:SSE2)
set(LLAMA_RUNTIME_PROFILE "baseline" CACHE STRING "Locked CPU profile")
set_property(CACHE LLAMA_RUNTIME_PROFILE PROPERTY STRINGS baseline avx2)
if(LLAMA_RUNTIME_PROFILE STREQUAL "baseline")
    set(profile_avx OFF)
elseif(LLAMA_RUNTIME_PROFILE STREQUAL "avx2")
    set(profile_avx ON)
else()
    message(FATAL_ERROR "Unknown LLAMA_RUNTIME_PROFILE")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/source-lock.json" profile_lock)
string(JSON profile_version GET "${profile_lock}" profiles ${LLAMA_RUNTIME_PROFILE} version)
string(JSON profile_driver GET "${profile_lock}" profiles ${LLAMA_RUNTIME_PROFILE} driver_requirements 0)
configure_file(component_profile.h.in component_profile.h @ONLY)
target_include_directories(llama-runtime PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_STATIC ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_SSE42 ON CACHE BOOL "" FORCE)
set(GGML_BMI2 ON CACHE BOOL "" FORCE)
set(GGML_AVX ${profile_avx} CACHE BOOL "" FORCE)
set(GGML_AVX2 ${profile_avx} CACHE BOOL "" FORCE)
set(GGML_FMA ${profile_avx} CACHE BOOL "" FORCE)
set(GGML_F16C ${profile_avx} CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_RPC OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_USE_PREBUILT_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_CONNECT OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_NUMBER 12182 CACHE STRING "" FORCE)
set(LLAMA_BUILD_COMMIT fd27b2034 CACHE STRING "" FORCE)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "" FORCE)
if(NOT IS_DIRECTORY "${LLAMA_SOURCE_DIR}")
    message(FATAL_ERROR "Verified LLAMA_SOURCE_DIR required")
endif()
add_subdirectory("${LLAMA_SOURCE_DIR}" upstream)
