cmake_minimum_required(VERSION 3.24)
project(fengshen_whisper_batch LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(WHISPER_SOURCE_DIR "" CACHE PATH "Verified whisper.cpp 4979e04")
set(JSON_HEADER "" CACHE FILEPATH "Verified json.hpp")
if(NOT EXISTS "${WHISPER_SOURCE_DIR}/include/whisper.h" OR NOT EXISTS "${JSON_HEADER}")
  message(FATAL_ERROR "Offline pinned dependencies required")
endif()
file(SHA256 "${JSON_HEADER}" JSON_SHA)
if(NOT JSON_SHA STREQUAL "aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63")
  message(FATAL_ERROR "JSON hash mismatch")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
foreach(option IN ITEMS BUILD_SHARED_LIBS WHISPER_BUILD_TESTS WHISPER_BUILD_EXAMPLES WHISPER_BUILD_SERVER WHISPER_CURL WHISPER_FFMPEG WHISPER_COREML WHISPER_OPENVINO WHISPER_CCACHE GGML_NATIVE GGML_CUDA GGML_HIP GGML_VULKAN GGML_METAL GGML_SYCL GGML_OPENCL GGML_BLAS GGML_RPC GGML_CANN GGML_MUSA GGML_OPENMP GGML_CCACHE GGML_BACKEND_DL)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "" FORCE)
set(FETCHCONTENT_UPDATES_DISCONNECTED ON CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper-build EXCLUDE_FROM_ALL)
get_filename_component(JSON_DIR "${JSON_HEADER}" DIRECTORY)
add_executable(whisper-batch main.cpp)
add_executable(whisper-batch-test batch_test.cpp)
foreach(target IN ITEMS whisper-batch whisper-batch-test)
  target_include_directories(${target} PRIVATE "${JSON_DIR}")
  target_link_libraries(${target} PRIVATE whisper)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
  endif()
endforeach()
if(MSVC)
  target_compile_options(whisper-batch-test PRIVATE /UNDEBUG)
else()
  target_compile_options(whisper-batch-test PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME whisper-batch-contract COMMAND whisper-batch-test)
